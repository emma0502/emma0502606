\section{Model}\whenintegrated{\label{model}}
\whenintegrated{\label{sec:model}} % integrated doc: SST for labels

A cost-minimizing government chooses how to allocate a given stock of
debt across three creditor sectors---domestic non-banks, foreigners,
and the banking sector. The three sectors differ in \emph{how}
placement becomes costly at the margin, and those differences produce
a tractable pecking order that we take to the data.

%-------------------------------------------------------------
\subsection{Setting}
%-------------------------------------------------------------

A government has a given amount of outstanding debt $D>0$ and chooses
non-negative holdings $(d_h,d_f,d_b)$ satisfying
\begin{equation}\label{eq:budget}
  d_h + d_f + d_b = D,
\end{equation}
where $d_h$ is non-bank holdings, $d_f$ is foreign holdings, and $d_b$
is banking (central bank plus private banks).

\paragraph{Every placement is costly.}
\begin{itemize}
  \item \textbf{Non-banks and foreigners} lend at interest rates
        $r_h(d_h,S_h)$ and $r_f(d_f)$; the government's cash cost of
        placement is the interest bill $r_h\,d_h + r_f\,d_f$.
  \item \textbf{Banking} is not summarised by an interest rate:
        placing $d_b$ with the banking system carries a separate
        welfare (quasi-fiscal) cost $\Psi(d_b)$---inflation,
        monetary/fiscal trade-offs, financial-stability risk.
\end{itemize}

\paragraph{Structural size.}
$S_h>0$ is the \emph{country-specific} non-bank sector size (capacity).

\paragraph{Planner problem.}
Total debt $D>0$ is given. Choose $d_h,d_f,d_b\ge 0$ to minimise total
placement cost subject to the budget constraint:
\begin{equation}\label{eq:planner}
  \min_{d_h,d_f,d_b}\;
    r_h(d_h,S_h)\,d_h + r_f(d_f)\,d_f + \Psi(d_b)
  \quad\text{s.t.}\quad d_h + d_f + d_b = D.
\end{equation}

%-------------------------------------------------------------
\subsection{The three sectors: how placement becomes costly}
%-------------------------------------------------------------

Marginal cost rises with the sector's \emph{degree of saturation}.
What differs is \emph{how} saturation is measured.

\paragraph{Non-banks.}
\begin{equation}\label{eq:nonbank-rate}
  r_h(d_h,S_h) = \bar r_h + \gamma_h\,
    \frac{d_h/S_h}{\bar\omega - d_h/S_h}.
\end{equation}
Saturation is the \emph{relative amount} $d_h/S_h$. The rate stays
near $\bar r_h$ when the sector is far from its cap $\bar\omega S_h$
and diverges as $d_h\to\bar\omega S_h$. Larger $S_h$ means more
headroom and later saturation. $S_h$ differs across countries.

\paragraph{Foreigners.}
\begin{equation}\label{eq:foreign-rate}
  r_f(d_f) = \bar r_f + \gamma_f\,
    \frac{d_f}{\bar D_f - d_f}.
\end{equation}
Saturation is the level of foreign exposure $d_f$. The rate diverges
as $d_f\to\bar D_f$ (a no-default ceiling). $\bar D_f$ is
\emph{common} across countries.

\paragraph{Banking.}
\begin{equation}\label{eq:banking-cost}
  \Psi(d_b),\qquad \Psi'(d_b)>0,\quad \Psi''(d_b)>0.
\end{equation}
There is no hard ceiling, but placement carries a welfare cost that
is increasing and convex.

%-------------------------------------------------------------
\subsection{Equilibrium: who absorbs the next dollar?}
%-------------------------------------------------------------

At the optimum, the planner equates the marginal cost of placing one
more dollar of debt across all three sectors:
\begin{equation}\label{eq:foc}
  MC_h(d_h;S_h) \;=\; MC_f(d_f) \;=\; MC_b(d_b) \;=\; \lambda(D),
  \qquad d_h + d_f + d_b = D.
\end{equation}
If one sector were cheaper at the margin, the planner would shift
toward it.

\paragraph{How the equilibrium plays out as $D$ grows.}
\begin{itemize}
  \item \textbf{Low $D$.} Non-banks and foreigners are far from their
        limits; $MC_h$ and $MC_f$ are low and they absorb most new
        debt. Banking is costlier at the margin and absorbs little.
  \item \textbf{Rising $D$.} Non-banks and foreigners approach their
        limits; $MC_h$ and $MC_f$ rise sharply and their absorption
        share shrinks. Banking still has room and starts picking up.
  \item \textbf{High $D$.} Non-banks and foreigners are near their
        caps; $MC_h, MC_f \to \infty$ and they absorb almost nothing
        at the margin. Banking is the only sector with a finite
        marginal cost and absorbs nearly all new debt.
\end{itemize}

%-------------------------------------------------------------
\subsection{Model predictions}
%-------------------------------------------------------------

\begin{enumerate}
  \item \textbf{Every sector absorbs more debt as $D$ grows.}
        Upward-sloping $MC_j$ implies that as $D$ rises, all three
        margins rise together and every $d_j^\ast$ rises.
  \item \textbf{Larger non-bank sectors absorb more debt at any given
        $D$.} A bigger $S_h$ means a less saturated non-bank sector
        at any $d_h$, a lower $MC_h$ at the margin, and so the
        planner allocates more to non-banks.
  \item \textbf{As $D$ grows, non-banks and foreigners absorb a
        smaller share of each extra dollar; their marginal shares go
        to zero. Under symmetry, the paths of $d_h^\ast$ and
        $d_f^\ast$ are strictly concave in $D$.} Both approach caps,
        so $MC_h$ and $MC_f$ rise and diverge, and the planner shifts
        each extra dollar toward banking.
  \item \textbf{Banking takeover at high $D$.} With non-banks and
        foreigners near their caps, banking is the only sector with
        room, and its share of new issuance rises sharply.
\end{enumerate}

%-------------------------------------------------------------
\subsection{Mechanics I: three phases of absorption}
%-------------------------------------------------------------

\noindent\textbf{One country, $S_h=150$.} Baseline calibration
$\bar r_h=\bar r_f=\bar r_b=1$, $\gamma_h=\gamma_f=0.02$,
$\bar\omega=0.30$, $\psi=0.04$, $\bar D_f=55$. Figure~\ref{fig:phases}
shades the three equilibrium phases, defined by banking's marginal
share $s_b \equiv d d_b^\ast/dD$.

\begin{figure}[h!]
  \centering
  \includegraphics[width=0.85\linewidth]{\latexroot/Figures/absorption_phases.png}
  \caption{Three phases of debt absorption for a representative
    country ($S_h=150$).}
  \label{fig:phases}
\end{figure}

\begin{itemize}
  \item \textbf{Phase 1.} Non-banks and foreigners absorb; banking
        flat.
  \item \textbf{Phase 2.} Non-banks and foreigners saturate; banking
        starts picking up.
  \item \textbf{Phase 3.} Banking takes over. The transition happens
        at $D\approx \bar\omega S_h+\bar D_f=100$, matching the
        empirical kink at $100\%$ of GDP documented in
        Section~\ref{sec:empirical}.
\end{itemize}

%-------------------------------------------------------------
\subsection{Mechanics II: heterogeneous non-bank sector}
%-------------------------------------------------------------

\noindent\textbf{Comparative statics in $S_h$}
($S_h\in\{30,70,110,150,200\}$; $\bar D_f$, $\bar\omega$,
$\gamma_h$, $\gamma_f$, $\psi$ fixed at the baseline).

\begin{figure}[h!]
  \centering
  \includegraphics[width=0.95\linewidth]{\latexroot/Figures/pecking_order.png}
  \caption{Pecking order across countries with different non-bank
    sector sizes $S_h$.}
  \label{fig:pecking-order}
\end{figure}

\paragraph{Pecking order.}
\begin{itemize}
  \item \textbf{Non-banks:} five different saturation levels, one per
        country, at $\bar\omega S_h$.
  \item \textbf{Foreigners:} essentially one curve---all countries
        approach the \emph{same} ceiling $\bar D_f$, differing only in
        how fast they get there.
  \item \textbf{Banking:} once total debt is large enough, banking
        takes over for all countries.
\end{itemize}
